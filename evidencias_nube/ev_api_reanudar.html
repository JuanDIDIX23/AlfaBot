43  /**
44   * Reanuda una evaluación guardada: reconstruye el estado probabilístico y los gránulos de las
45   * respuestas ya dadas. Es lo que permite que el motor funcione como servicio sin memoria entre
46   * peticiones. Devuelve null si la evaluación no existe.
47   */
48  async reanudarSesion(evaluacionId: number): Promise<{ sesion: SesionDiagnostico; completada: boolean } | null> {
49    const estado = await this.evaluaciones.estadoEvaluacion(evaluacionId);
50    if (estado === null) return null;
51
52    const hipotesis = await this.conocimiento.hipotesis();
53    if (estado.creencias.length !== hipotesis.length) {
54      throw new Error(`La evaluación ${evaluacionId} no tiene un estado probabilístico completo`);
55    }
56    const probabilidades = new Map(estado.creencias.map((c) => [c.hipotesisId, c.probabilidad]));
57    const creencias = hipotesis.map((h) => {
58      const p = probabilidades.get(h.id);
59      if (p === undefined) throw new Error(`Falta la probabilidad de la hipótesis ${h.codigo}`);
60      return new Creencia(h, p);
61    });
62
63    const granulos = [];
64    for (const r of estado.respuestas) granulos.push(await this.conocimiento.granuloDeRama(r.ramaId));
65
66    const sesion = new SesionDiagnostico(
67      estado.usuarioId,
68      evaluacionId,
69      creencias,
70      {
71        conocimiento: this.conocimiento,
72        evaluaciones: this.evaluaciones,
73        motor: this.motor,
74        selector: this.selector,
75        criterio: this.criterio,
76        generador: this.generador,
77      },
78      { nodosRespondidos: estado.respuestas.map((r) => r.nodoId), granulos },
79    );
80    return { sesion, completada: estado.estado === "completada" };
81  }
