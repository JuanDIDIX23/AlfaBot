61  async responder(idTexto: string, cuerpo: unknown): Promise<RespuestaHttp> {
62    const id = this.validarId(idTexto);
63    const ramaId = this.validarRama(cuerpo);
64
65    const reanudada = await this.servicio.reanudarSesion(id);
66    if (reanudada === null) throw new ErrorHttp(404, "No existe esa evaluación");
67    const { sesion, completada } = reanudada;
68    if (completada) throw new ErrorHttp(409, "Esa evaluación ya terminó");
69
70    const pregunta = await sesion.siguientePregunta();
71    if (pregunta === null) {
72      await sesion.finalizar(); // evaluación que quedó sin cerrar: se cierra y se informa
73      throw new ErrorHttp(409, "Esa evaluación ya terminó");
74    }
75    if (!pregunta.opciones.some((o) => o.ramaId === ramaId)) {
76      throw new ErrorHttp(400, "Esa respuesta no corresponde a la pregunta actual");
77    }
78
79    const paso = await sesion.responder(ramaId);
80    const estado = await this.estadoDe(sesion, false);
81    return {
82      estado: 200,
83      cuerpo: {
84        ...estado,
85        regla: paso.rama && paso.reglaCodigo ? { codigo: paso.reglaCodigo, condicion: paso.rama.condicion } : null,
86      },
87    };
88  }
